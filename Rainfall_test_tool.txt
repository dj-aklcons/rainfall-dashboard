    <script>
        const config = {
            baseUrl: "http://aklc.hydrotel.co.nz:8080/KiWIS/KiWIS",
            format: "dajson"
        };

        const siteMap = {
            "648719": "Albert Park (Central)",
            "647722": "Keeling Road (Waitakere)",
            "648612": "Wairau (Takapuna)",
            "649940": "Sports Bowl (Manukau)"
        };

        function setFormat(f) {
            config.format = f;
            document.querySelectorAll('.format-btn').forEach(btn => {
                if (btn.getAttribute('data-value') === f) {
                    btn.className = "format-btn px-3 py-2 text-sm border rounded-lg bg-blue-50 border-blue-500 text-blue-700 font-medium";
                } else {
                    btn.className = "format-btn px-3 py-2 text-sm border rounded-lg bg-white border-slate-300 text-slate-600";
                }
            });
            updateUrl();
        }

        function updateUrl() {
            const siteId = document.getElementById('siteId').value;
            const pVal = document.getElementById('periodValue').value;
            const pUnit = document.getElementById('periodUnit').value;
            const tz = document.getElementById('timezone').value;
            
            const period = `P${pVal}${pUnit}`;
            const ts_id = `${siteId}~Rainfall.HOURTOT`;
            
            const params = new URLSearchParams({
                service: "kisters",
                type: "queryServices",
                request: "getTimeseriesValues",
                datasource: "0",
                format: config.format,
                ts_id: ts_id,
                period: period,
                returnfields: "Timestamp,Value,Quality Code",
                timezone: tz
            });

            const finalUrl = `${config.baseUrl}?${params.toString()}`;
            
            // Update UI
            document.getElementById('urlOutput').innerText = finalUrl;
            document.getElementById('summarySite').innerText = siteMap[siteId];
            document.getElementById('summaryPeriod').innerText = period;
            document.getElementById('summaryFormat').innerText = config.format;
        }

        function copyUrl() {
            const url = document.getElementById('urlOutput').innerText;
            const el = document.createElement('textarea');
            el.value = url;
            document.body.appendChild(el);
            el.select();
            document.execCommand('copy');
            document.body.removeChild(el);
            
            const toast = document.getElementById('copyToast');
            toast.style.opacity = '1';
            setTimeout(() => toast.style.opacity = '0', 2000);
        }

        function testUrl() {
            const url = document.getElementById('urlOutput').innerText;
            window.open(url, '_blank');
        }

        // Add event listeners
        document.getElementById('siteId').addEventListener('change', updateUrl);
        document.getElementById('periodValue').addEventListener('input', updateUrl);
        document.getElementById('periodUnit').addEventListener('change', updateUrl);
        document.getElementById('timezone').addEventListener('change', updateUrl);

        // Initial build
        window.onload = updateUrl;
    </script>
